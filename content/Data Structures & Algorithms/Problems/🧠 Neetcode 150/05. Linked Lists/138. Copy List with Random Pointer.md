## 🔍 Problem Understanding

- we are given a linked list of nodes where each node has the following member variables:
```
int val;
Node* next;
Node* random;
```

- the random member can point to any node in the list or null
- need to construct a **deep copy** where there should be exactly n brand new nodes 
- the next and random pointers should be assigned correctly to these new nodes

## 💻 Implementation of Solution

```cpp
/*
// Definition for a Node.
class Node {
public:
    int val;
    Node* next;
    Node* random;
    
    Node(int _val) {
        val = _val;
        next = NULL;
        random = NULL;
    }
};
*/

class Solution {
public:
    Node* copyRandomList(Node* head) {
        if(!head) return nullptr;

        unordered_map<Node*, Node*> map;

        Node* curr = head;
        while(curr) {
            map[curr] = new Node(curr->val);
            curr = curr->next;
        }

        curr = head;
        while(curr) {
            map[curr]->next = map[curr->next];
            map[curr]->random = map[curr->random];
            curr = curr->next;
        }

        return map[head];
    }
};
```
