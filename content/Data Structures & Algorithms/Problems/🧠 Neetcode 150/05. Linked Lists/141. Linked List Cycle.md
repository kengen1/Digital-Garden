## 💻 Implementation of Solution

```cpp
class Solution {
public:
    bool hasCycle(ListNode *head) {

     unordered_set<ListNode*> set;

     ListNode* curr = head;

     while(curr != nullptr) {
        if(set.find(curr) != set.end()) {
            return true;
        } else {
            set.insert(curr);
            curr = curr->next;
        }
     }
     return false;   
    }
};
```
